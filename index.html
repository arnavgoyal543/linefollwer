from machine import Pin, PWM import ubluetooth, struct # Motor pins AIN1 = Pin(16, Pin.OUT) AIN2 = Pin(17, Pin.OUT) PWMA = PWM(Pin(4), freq=1000) BIN1 = Pin(18, Pin.OUT) BIN2 = Pin(19, Pin.OUT) PWMB = PWM(Pin(5), freq=1000) STBY = Pin(23, Pin.OUT) STBY.value(1) # BLE setup ble = ubluetooth.BLE() ble.active(True) UART_UUID = ubluetooth.UUID(0xFFE0) TXRX_UUID = ubluetooth.UUID(0xFFE1) UART_SERVICE = ( UART_UUID, (ubluetooth.FLAG_WRITE | ubluetooth.FLAG_READ,), ) handle = None def motor_control(m1, m2): # Motor A if m1 > 0: AIN1.value(1); AIN2.value(0) elif m1 < 0: AIN1.value(0); AIN2.value(1) else: AIN1.value(0); AIN2.value(0) PWMA.duty(abs(m1)) # Motor B if m2 > 0: BIN1.value(1); BIN2.value(0) elif m2 < 0: BIN1.value(0); BIN2.value(1) else: BIN1.value(0); BIN2.value(0) PWMB.duty(abs(m2)) def bt_irq(event, data): global handle if event == 1: # Central connected handle = data[0] elif event == 3: # Write event buffer = ble.gatts_read(data[1]) if len(buffer) == 2: m1, m2 = buffer[0], buffer[1] motor_control(m1, m2) ble.irq(bt_irq) ((handle,),) = ble.gatts_register_services((UART_SERVICE,)) ble.gap_advertise(100, b'\x02\x01\x06\x03\x03\xe0\xff\x0b\x09ESP32Bot')
